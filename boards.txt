# OPEN-pLC H7
OPEN-PLC.name=OPEN-PLC
# USB ID of the board's CDC port, so the IDE port list names the board (R3-25).
# 0483:5740 is ST's generic CDC ID; a dedicated PID only changes these two lines.
OPEN-PLC.upload_port.0.vid=0x0483
OPEN-PLC.upload_port.0.pid=0x5740

# Fallback so the upload patterns resolve without a forceflash selection.
OPEN-PLC.upload.force_flag=
OPEN-PLC.build.core=arduino
OPEN-PLC.build.board= GenH7
# The one place this release's version is written. It reaches the firmware as
# OPENPLC_FW_VERSION, the third field of the identity string a board answers
# discovery with. It says which release built the firmware; it is not a version
# for the sketch, and nothing compares it. Keep it dotted, each field 0-255.
OPEN-PLC.build.fw_version=0.1.3
# HAL_RNG_MODULE_ENABLED: the challenge nonce comes from the RNG peripheral
# (libraries/OpenPLC_IAP/src/iap_auth.c, decision 66). RNG is not in the opt-in
# list in cores/arduino/stm32/stm32yyxx_hal_conf.h, so it has to be defined here.
OPEN-PLC.build.st_extra_flags=-DCORE_CM7 -DHAL_RNG_MODULE_ENABLED -D{build.product_line} {build.enable_usb} {build.xSerial} '-DOPENPLC_FW_VERSION="{build.fw_version}"'
OPEN-PLC.build.cmsis_lib_gcc=arm_cortexM7lfsp_math
OPEN-PLC.build.fpu=-mfpu=fpv4-sp-d16
OPEN-PLC.build.float-abi=-mfloat-abi=hard
OPEN-PLC.build.series=STM32H7xx
OPEN-PLC.build.mcu=cortex-m7
OPEN-PLC.upload.maximum_size=0
OPEN-PLC.upload.maximum_data_size=0

# PLC H743IIKx
# upload.maximum_size must equal IAP_APP_MAX_SIZE in the bootloader project
# (Core/Inc/usbd_cdc_flash.h: IAP_STATE_SECTOR_ADDR - IAP_APP_ADDRESS). The app
# region starts after the bootloader's own first sector (0x08020000) and ends
# before the last sector, reserved for firmware metadata/event log
# (IAP_STATE_SECTOR_ADDR) -- both are off-limits to the application. 1966080
# (2MB minus one sector) counted only the state sector and missed the
# bootloader's own sector; see $PROD/docs/tables/DECISIONS.md.
OPEN-PLC.menu.pnum.PLC_H743=PLC H743
OPEN-PLC.menu.pnum.PLC_H743.upload.maximum_size=1835008
OPEN-PLC.menu.pnum.PLC_H743.upload.maximum_data_size=524288
OPEN-PLC.menu.pnum.PLC_H743.build.board=PLC_H743
OPEN-PLC.menu.pnum.PLC_H743.build.product_line=STM32H743xx
OPEN-PLC.menu.pnum.PLC_H743.build.variant_h=variant_{build.board}.h
OPEN-PLC.menu.pnum.PLC_H743.build.variant=STM32H7xx/H743
OPEN-PLC.menu.pnum.PLC_H743.build.peripheral_pins=-DCUSTOM_PERIPHERAL_PINS
OPEN-PLC.menu.pnum.PLC_H743.build.ldscript=ldscript.ld
# USB_CDC_TRANSMIT_TIMEOUT: upstream 3 ms drops lines while a terminal is open ($PROD/docs/tables/DECISIONS.md 124)
OPEN-PLC.menu.pnum.PLC_H743.build.extra_flags=-DUSB_CDC_TRANSMIT_TIMEOUT=20 -DLAN8742A_PHY_ADDRESS=0 -DETH_RXBUFNB=4 -DETH_TXBUFNB=4 -DETH_RX_BUF_SIZE=1536 -DETH_TX_BUF_SIZE=1536 -DOPENPLC_UDP_SERVER_AUTOSTART -DD_CACHE_DISABLED -DADC_SAMPLINGTIME=ADC_SAMPLETIME_387CYCLES_5 -DADC_CLOCK_DIV=ADC_CLOCK_ASYNC_DIV4

# USB connectivity. CDC (no generic 'Serial') is for Renode, where Serial is
# read on UART4 ($PROD/maps/sim-coverage/SIM-01-findings.md).
OPEN-PLC.menu.usb.CDCgen=CDC (generic 'Serial' supersede U(S)ART)
OPEN-PLC.menu.usb.CDCgen.build.enable_usb={build.usb_flags} -DUSBD_USE_CDC
OPEN-PLC.menu.usb.CDC=CDC (no generic 'Serial')
OPEN-PLC.menu.usb.CDC.build.enable_usb={build.usb_flags} -DUSBD_USE_CDC -DDISABLE_GENERIC_SERIALUSB
